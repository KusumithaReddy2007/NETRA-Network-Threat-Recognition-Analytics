import { useEffect, useMemo, useState } from "react";
import {
  Activity,
  BarChart3,
  Bell,
  Bot,
  ChevronDown,
  Eye,
  FileBarChart,
  Home,
  Menu,
  Moon,
  Network,
  Radar,
  Search,
  Settings,
  ShieldAlert,
} from "lucide-react";
import { AnimatePresence, motion } from "framer-motion";

import {
  getAnalytics,
  replayTraffic,
  healthCheck,
} from "./services/api";

import DashboardPage from "./pages/DashboardPage";
import LiveMonitoringPage from "./pages/LiveMonitoringPage";
import ThreatsPage from "./pages/ThreatsPage";
import AnalyticsPage from "./pages/AnalyticsPage";
import FlowsPage from "./pages/FlowsPage";
import AlertsPage from "./pages/AlertsPage";
import ReportsPage from "./pages/ReportsPage";
import ModelPage from "./pages/ModelPage";

import "./App.css";

const NAVIGATION = [
  { label: "Dashboard", icon: Home },
  { label: "Live Monitoring", icon: Radar },
  { label: "Threats", icon: ShieldAlert },
  { label: "Analytics", icon: BarChart3 },
  { label: "Flows", icon: Network },
  { label: "Alerts", icon: Bell },
  { label: "Reports", icon: FileBarChart },
  { label: "Model", icon: Bot },
  { label: "Settings", icon: Settings },
];

function App() {
  const [activePage, setActivePage] =
    useState("Dashboard");

  const [mobileOpen, setMobileOpen] =
    useState(false);

  const [analytics, setAnalytics] =
    useState(null);

  const [alerts, setAlerts] = useState([]);

  const [online, setOnline] =
    useState(false);

  const [booting, setBooting] =
    useState(true);

  const [dataLoading, setDataLoading] =
    useState(true);

  const [replaying, setReplaying] =
    useState(false);

  const [error, setError] =
    useState("");

  async function loadAnalytics() {
    try {
      setDataLoading(true);
      setError("");

      const result =
        await getAnalytics(100);

      if (result?.analytics) {
        setAnalytics(result.analytics);
        setOnline(true);
      }
    } catch (err) {
      console.error(
        "NETRA analytics error:",
        err
      );

      setOnline(false);

      setError(
        err?.response?.data?.detail ||
          err?.message ||
          "Analytics service unavailable"
      );
    } finally {
      setDataLoading(false);
    }
  }

  async function runReplay() {
    if (replaying) return;

    try {
      setReplaying(true);
      setError("");

      const result =
        await replayTraffic(100, 0);

      if (
        Array.isArray(result?.alerts)
      ) {
        setAlerts(result.alerts);
      }

      await loadAnalytics();
    } catch (err) {
      console.error(
        "NETRA replay error:",
        err
      );

      setError(
        err?.response?.data?.detail ||
          err?.message ||
          "Replay failed"
      );
    } finally {
      setReplaying(false);
    }
  }

  useEffect(() => {
    const timer = setTimeout(() => {
      setBooting(false);
    }, 1700);

    healthCheck()
      .then(() => setOnline(true))
      .catch(() => setOnline(false));

    loadAnalytics();

    return () =>
      clearTimeout(timer);
  }, []);

  const threatTypes = useMemo(() => {
    return Object.entries(
      analytics?.threat_types || {}
    )
      .map(([name, value]) => ({
        name,
        value: Number(value),
      }))
      .sort(
        (a, b) => b.value - a.value
      );
  }, [analytics]);

  const severity =
    analytics?.severity_breakdown || {};

  const page = renderPage({
    activePage,
    analytics,
    alerts,
    threatTypes,
    severity,
    replaying,
    dataLoading,
    error,
    runReplay,
  });

  if (booting) {
    return <BootScreen />;
  }

  return (
    <div className="netra-app">
      <CyberAtmosphere />

      <AnimatePresence>
        {mobileOpen && (
          <motion.div
            className="mobile-overlay"
            initial={{ opacity: 0 }}
            animate={{ opacity: 1 }}
            exit={{ opacity: 0 }}
            onClick={() =>
              setMobileOpen(false)
            }
          />
        )}
      </AnimatePresence>

      <aside
        className={`sidebar ${
          mobileOpen
            ? "sidebar-open"
            : ""
        }`}
      >
        <div className="brand">
          <div className="brand-mark">
            <Eye size={24} />
            <span className="brand-ring" />
          </div>

          <div>
            <div className="brand-name">
              NETRA
            </div>

            <div className="brand-subtitle">
              NETWORK THREAT RECOGNITION
            </div>
          </div>
        </div>

        <div className="sidebar-label">
          OPERATIONS
        </div>

        <nav className="navigation">
          {NAVIGATION.map((item) => {
            const Icon = item.icon;

            const active =
              activePage === item.label;

            return (
              <button
                key={item.label}
                className={`nav-item ${
                  active
                    ? "active"
                    : ""
                }`}
                onClick={() => {
                  setActivePage(
                    item.label
                  );
                  setMobileOpen(false);
                }}
              >
                <span className="nav-icon">
                  <Icon size={18} />
                </span>

                <span className="nav-text">
                  {item.label}
                </span>

                {item.label ===
                  "Threats" &&
                  analytics?.threat_flows >
                    0 && (
                    <span className="nav-badge">
                      {analytics.threat_flows}
                    </span>
                  )}

                {item.label ===
                  "Alerts" &&
                  alerts.length >
                    0 && (
                    <span className="nav-badge">
                      {alerts.length}
                    </span>
                  )}

                {active && (
                  <span className="nav-glow" />
                )}
              </button>
            );
          })}
        </nav>

        <div className="sidebar-spacer" />

        <div className="system-panel">
          <div className="panel-header">
            <span>
              SYSTEM STATUS
            </span>

            <Activity size={15} />
          </div>

          <StatusRow
            label="AI ENGINE"
            value={
              online
                ? "ONLINE"
                : "OFFLINE"
            }
            online={online}
          />

          <StatusRow
            label="STREAM PROCESSOR"
            value={
              online
                ? "ONLINE"
                : "OFFLINE"
            }
            online={online}
          />

          <StatusRow
            label="MODEL"
            value="ACTIVE"
            online
          />

          <StatusRow
            label="DATASET"
            value="CIC-IDS2017"
          />

          <StatusRow
            label="MODE"
            value="PASSIVE"
          />
        </div>

        <div className="sidebar-foot">
          <span>NETRA CORE</span>
          <span>v1.0.0</span>
        </div>
      </aside>

      <div className="main-area">
        <header className="topbar">
          <div className="topbar-left">
            <button
              className="mobile-menu"
              onClick={() =>
                setMobileOpen(
                  !mobileOpen
                )
              }
            >
              <Menu size={22} />
            </button>

            <div className="breadcrumb">
              <span>
                NETRA
              </span>
              <i>/</i>
              <strong>
                {activePage}
              </strong>
            </div>
          </div>

          <div className="global-search">
            <Search size={17} />

            <input
              placeholder="Search threats, IPs, flows..."
            />

            <kbd>
              ⌘ K
            </kbd>
          </div>

          <div className="topbar-right">
            <div className="connection">
              <span />
              {online
                ? "SYSTEM ONLINE"
                : "SYSTEM OFFLINE"}
            </div>

            <button className="top-icon">
              <Bell size={19} />

              {alerts.length >
                0 && (
                <span className="notification-dot">
                  {alerts.length >
                  99
                    ? "99+"
                    : alerts.length}
                </span>
              )}
            </button>

            <button className="top-icon">
              <Moon size={18} />
            </button>

            <div className="user-block">
              <div className="user-avatar">
                A
              </div>

              <div>
                <strong>
                  Admin
                </strong>

                <span>
                  Security Operator
                </span>
              </div>

              <ChevronDown
                size={16}
              />
            </div>
          </div>
        </header>

        <main className="content">
          <AnimatePresence mode="wait">
            <motion.div
              key={activePage}
              initial={{
                opacity: 0,
                y: 10,
              }}
              animate={{
                opacity: 1,
                y: 0,
              }}
              exit={{
                opacity: 0,
                y: -8,
              }}
              transition={{
                duration: 0.22,
              }}
            >
              {page}
            </motion.div>
          </AnimatePresence>

          <footer className="app-footer">
            <div>
              NETRA SECURITY ENGINE
              <span>•</span>
              PASSIVE MONITORING
            </div>

            <div>
              READ-ONLY DETECTION MODE
            </div>
          </footer>
        </main>
      </div>
    </div>
  );
}

function renderPage({
  activePage,
  analytics,
  alerts,
  threatTypes,
  severity,
  replaying,
  dataLoading,
  error,
  runReplay,
}) {
  switch (activePage) {
    case "Live Monitoring":
      return (
        <LiveMonitoringPage
          analytics={analytics}
          alerts={alerts}
          runReplay={runReplay}
          replaying={replaying}
        />
      );

    case "Threats":
      return (
        <ThreatsPage
          alerts={alerts}
        />
      );

    case "Analytics":
      return (
        <AnalyticsPage
          analytics={analytics}
        />
      );

    case "Flows":
      return (
        <FlowsPage
          analytics={analytics}
        />
      );

    case "Alerts":
      return (
        <AlertsPage
          alerts={alerts}
        />
      );

    case "Reports":
      return (
        <ReportsPage
          analytics={analytics}
        />
      );

    case "Model":
      return (
        <ModelPage
          analytics={analytics}
        />
      );

    case "Settings":
      return (
        <SettingsPage />
      );

    default:
      return (
        <DashboardPage
          analytics={analytics}
          alerts={alerts}
          threatTypes={threatTypes}
          severity={severity}
          replaying={replaying}
          dataLoading={dataLoading}
          error={error}
          runReplay={runReplay}
        />
      );
  }
}

function StatusRow({
  label,
  value,
  online = false,
}) {
  return (
    <div className="status-row">

      <strong
        className={
          online ? "online" : ""
        }
      >
        {online && <i />}
        {value}
      </strong>
    </div>
  );
}

import axios from "axios";

const API_BASE_URL = "http://127.0.0.1:8000";

const api = axios.create({
  baseURL: API_BASE_URL,
  timeout: 120000,
  headers: {
    "Content-Type": "application/json",
  },
});

export async function healthCheck() {
  const response = await api.get("/health");
  return response.data;
}

export async function getAnalytics(maxFlows = 100) {
  const response = await api.get("/analytics/summary", {
    params: {
      max_flows: maxFlows,
    },
  });

  return response.data;
}

export async function replayTraffic(maxFlows = 100, delay = 0) {
  const response = await api.post("/stream/replay", null, {
    params: {
      max_flows: maxFlows,
      delay,
    },
  });

  return response.data;
}

export async function analyzeFlow(flowData) {
  const response = await api.post("/threat/analyze", flowData);
  return response.data;
}

export default api;

  if (!timestamp) return "--:--";

function formatNumber(value) {
  const number = Number(value);

  if (!Number.isFinite(number)) {
    return "0";
  }

  return Number.isInteger(number)
    ? String(number)
    : number.toFixed(2);
}

function formatKey(key) {
  return String(key)
    .replaceAll("_", " ")
    .replace(/\b\w/g, (letter) => letter.toUpperCase());
}

function formatEvidenceValue(value) {
  if (value === null || value === undefined) {
    return "N/A";
  }

  if (typeof value === "boolean") {
    return value ? "YES" : "NO";
  }

  if (typeof value === "number") {
    return Number.isInteger(value)
      ? String(value)
      : value.toFixed(3);
  }

  return String(value);
}

import { useMemo, useState } from "react";
import {
  AlertTriangle,
  Eye,
  Filter,
  Search,
  ShieldAlert,
  X,
} from "lucide-react";
import { motion, AnimatePresence } from "framer-motion";

function ThreatsPage({ alerts }) {
  const [filter, setFilter] = useState("ALL");
  const [search, setSearch] = useState("");
  const [selected, setSelected] = useState(null);

  const filtered = useMemo(() => {
    return alerts.filter((alert) => {
      const matchesFilter =
        filter === "ALL" ||
        alert.threat_class === filter;

      const text = `${alert.threat_class} ${alert.flow_id} ${alert.alert_id}`.toLowerCase();

      return (
        matchesFilter &&
        text.includes(search.toLowerCase())
      );
    });
  }, [alerts, filter, search]);

  const types = [
    "ALL",
    ...new Set(alerts.map((a) => a.threat_class)),
  ];

  return (
    <div className="special-page">
      <PageHeader />

      <div className="threat-center">
        <div className="threat-toolbar">
          <div className="filter-title">
            <ShieldAlert size={19} />
            <div>
              <strong>THREAT CENTER</strong>
              <span>
                {filtered.length} visible detections
              </span>
            </div>
          </div>

          <div className="threat-search">
            <Search size={16} />
            <input
              value={search}
              onChange={(e) =>
                setSearch(e.target.value)
              }
              placeholder="Search alert ID, flow ID, threat..."
            />
          </div>
        </div>

        <div className="filter-row">
          <Filter size={14} />

          {types.map((type) => (
            <button
              key={type}
              className={
                filter === type ? "filter-active" : ""
              }
              onClick={() => setFilter(type)}
            >
              {type.replaceAll("_", " ")}
            </button>
          ))}
        </div>

        <div className="threat-table-panel">
          <table className="investigation-table">
            <thead>
              <tr>
                <th>ALERT ID</th>
                <th>THREAT</th>
                <th>FLOW ID</th>
                <th>TIME</th>
                <th>SEVERITY</th>
                <th>CONFIDENCE</th>
                <th />
              </tr>
            </thead>

            <tbody>
              {filtered.map((alert, index) => (
                <motion.tr
                  key={
                    alert.alert_id || index
                  }
                  initial={{ opacity: 0 }}
                  animate={{ opacity: 1 }}
                >
                  <td className="mono-cell">
                    {alert.alert_id}
                  </td>

                  <td>
                    <span className="table-threat">
                      {alert.threat_class}
                    </span>
                  </td>

                  <td className="mono-cell">
                    {alert.flow_id}
                  </td>

                  <td>
                    {formatTime(alert.timestamp)}
                  </td>

                  <td>
                    <Severity
                      value={alert.severity}
                    />
                  </td>

                  <td className="confidence-cell">
                    {alert.confidence}%
                  </td>

                  <td>
                    <button
                      className="investigate-button"
                      onClick={() =>
                        setSelected(alert)
                      }
                    >
                      <Eye size={15} />
                      INVESTIGATE
                    </button>
                  </td>
                </motion.tr>
              ))}
            </tbody>
          </table>

          {!filtered.length && (
            <div className="threat-empty">
              <AlertTriangle size={24} />
              <strong>No matching detections</strong>
              <span>
                Try another threat filter or search.
              </span>
            </div>
          )}
        </div>
      </div>

      <AnimatePresence>
        {selected && (
          <motion.div
            className="detail-overlay"
            initial={{ opacity: 0 }}
            animate={{ opacity: 1 }}
            exit={{ opacity: 0 }}
            onClick={() => setSelected(null)}
          >
            <motion.aside
              className="detail-drawer"
              initial={{ x: 430 }}
              animate={{ x: 0 }}
              exit={{ x: 430 }}
              transition={{ type: "spring", damping: 27 }}
              onClick={(e) => e.stopPropagation()}
            >
              <button
                className="drawer-close"
                onClick={() => setSelected(null)}
              >
                <X size={18} />
              </button>

              <div className="drawer-eyebrow">
                SECURITY EVENT
              </div>

              <h2>{selected.threat_class}</h2>

              <Severity
                value={selected.severity}
              />

              <div className="drawer-grid">
                <Detail label="ALERT ID" value={selected.alert_id} />
                <Detail label="FLOW ID" value={selected.flow_id} />
                <Detail label="TIMESTAMP" value={selected.timestamp} />
                <Detail
                  label="CONFIDENCE"
                  value={`${selected.confidence}%`}
                />
              </div>

              <div className="evidence">
                <div className="evidence-heading">
                  <strong>SUPPORTING EVIDENCE</strong>
                  <span>MODEL SIGNALS</span>
                </div>

                {Object.entries(
                  selected.supporting_evidence || {}
                ).map(([key, value]) => (
                  <div className="evidence-row" key={key}>
                    <span>{key.replaceAll("_", " ")}</span>
                    <strong>{String(value)}</strong>
                  </div>
                ))}
              </div>
            </motion.aside>
          </motion.div>
        )}
      </AnimatePresence>
    </div>
  );
}

function PageHeader() {
  return (
    <div className="page-heading">
      <div>
        <div className="eyebrow">
          <span className="eyebrow-line" />
          THREAT INVESTIGATION
        </div>

        <h1>Threat Center</h1>

        <p>
          Investigate standardized NETRA security
          detections and their supporting evidence.
        </p>
      </div>
    </div>
  );
}

function Severity({ value }) {
  const text = String(value || "HIGH").toUpperCase();

  return (
    <span
      className={`severity-badge ${text.toLowerCase()}`}
    >
      {text}
    </span>
  );
}

function Detail({ label, value }) {
  return (
    <div className="drawer-detail">
      <span>{label}</span>
      <strong>{value || "—"}</strong>
    </div>
  );
}

function formatTime(timestamp) {
  if (!timestamp) return "--:--:--";

  try {
    return new Date(timestamp).toLocaleTimeString([], {
      hour12: false,
    });
  } catch {
    return "--:--:--";
  }
}

export default ThreatsPage;
